# dform.df, direction E.
#
#   dform plan
#   dform plan --set env=prod
#   dform apply

edition 2026

# Typed stack input with a default. The default is a @default contribution
# to input env; --set env=prod is a normal contribution and wins.
input env: enum("dev", "staging", "prod") = "staging"

provider fake { source = "./providers/fake" }

zone("us-test-1a")
zone("us-test-1b")

iam_need("app", "audit.write", "audit-log") where env == "prod"

# Settings are a resource of type `settings`, addressed by environment.
# A user schema for the open type gives typed, refined settings.
type settings {
  audit.sinks: set(string)
  db.backup_days: int check 1 <= db.backup_days <= 35
}

# Base layer at rank @default: every environment. Overrides below are the
# normal rank and win per leaf. This replaces three copies of the block.
settings e @default {
  network.main.vpc_net = inet("10.90.0.0/16")
  network.peer.vpc_net = inet("10.91.0.0/16")
  db = { backup_days: 3, multi_az: false }
  k8s = { private_api: false, nodepool: { min: 1, max: 3 } }
} where env_name(e)

env_name("dev")
env_name("staging")
env_name("prod")

settings staging {
  network.main.vpc_net = inet("10.50.0.0/16")
  network.peer.vpc_net = inet("10.60.0.0/16")
}

settings prod {
  network.main.vpc_net = inet("10.20.0.0/16")
  network.peer.vpc_net = inet("10.21.0.0/16")
  db = { backup_days: 14, multi_az: true }
  k8s = { private_api: true, nodepool: { min: 3, max: 10 } }
  audit.sinks += ["cloudwatch"] # Set path: unions with the policy pack's ["s3"]
}

use baseline
use stdlib_net

# Copies of components. Inputs are typed at the component; the body binds them.
instance network main {
  vpc_net = v
} where e = env, setting(e, "network.main.vpc_net", v)
instance network peer {
  vpc_net = v
} where e = env, setting(e, "network.peer.vpc_net", v)

instance database primary {
  backup_days = days
  multi_az = m_a
  subnet_ids = ids
} where {
  e = env
  setting(e, "db.backup_days", days)
  setting(e, "db.multi_az", m_a)
  output("network.main", "private_subnet_ids", ids)
}

instance kubernetes apps {
  private_api = api_
  nodepool_min = min
  nodepool_max = max
  subnet_ids = ids
} where {
  e = env
  setting(e, "k8s.private_api", api_)
  setting(e, "k8s.nodepool.min", min)
  setting(e, "k8s.nodepool.max", max)
  output("network.main", "private_subnet_ids", ids)
}

use identity

# Peering between network instances. Each instance exports its vpc address.
vpc_peer_inst("main", "peer")

vpc_peer(a_, b) where {
  vpc_peer_inst(i_a, i_b)
  output(network[i_a], "vpc", a_)
  output(network[i_b], "vpc", b)
}

# requester_vpc_id and accepter_vpc_id are fresh nulls at plan time. They are
# forwarded, never inspected, so this resource is definite and single-phase.
resource net.vpc_peering peer {
  requester_vpc_id = ref(net.vpc, a_, "id")
  accepter_vpc_id = ref(net.vpc, b, "id")
} where vpc_peer_inst(i_a, i_b), vpc_peer(a_, b), peer = format("peer-%s-%s", i_a, i_b)

# A bastion with an address carved from the main VPC. The setting is a
# constant, so inet_host is not stuck; only subnet_id carries a null.
resource compute.vm bastion {
  private_ip = inet.host(net_, 20)
  subnet_id = ref(net.subnet, network.main / private_a, "id")
} where e = env, setting(e, "network.main.vpc_net", net_)

# Stack-level policy. Ids are stable; messages are free text.
deny("prod_db_multi_az", "prod db must be multi_az", { db: db_ }) where {
  env == "prod"
  want("db.postgres", db_)
  not attr("db.postgres", db_, "multi_az", true)
}

deny("prod_k8s_private_api", "prod k8s must have private api", { cluster: c }) where {
  env == "prod"
  want(k8s.cluster, c)
  not attr(k8s.cluster, c, "private_api", true)
}
