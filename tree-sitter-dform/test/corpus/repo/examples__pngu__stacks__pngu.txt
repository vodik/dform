# pngu.df
#
# dform representation of ~/src/movableink/pngu/terraform.
#
# To run it, from this directory: README.md.
#
# Each env value is its own deployment with its own state:
# dform.state/pngu/env=prod/ never sees dev's resources. Each env is its own
# GCP project (settings below), so names may repeat across envs: isolated.
#
# Notes:
# - This file models the intent/config surface area of the Terraform stack.
# - External Terraform modules are represented as higher-level resources.
# - The monitoring dashboard JSON is read by the loader `json(path)` from a stub,
#   files/dashboard-pngu.json (see terraform/modules/monitoring/dashboard-pngu.json).

edition 2026

# The stack's key, typed: `dform plan pngu env=qa` is an error naming it.
key env: environment = "dev"
# The project and its networks, each environment's in a settings block
# below. A CIDR is text here, made an inet where it is used.
input gcp {
  project_id: string
  project_name: string
  region: string
  network_name: string
  regional_proxy_cidr: string
}
input gke {
  control_plane_cidr: string
  subnet_cidr: string
  pods_cidr: string
  services_cidr: string
  static_ips: any
}
# The rows are data/peerings.csv's, typed column by column as `decl peering`
# says: a row whose env is not one of these is an error naming its line.
input peering from csv("data/peerings.csv")

decl peering(env: environment, name: string, peer_network: string)

########################################
# Provider (configured by the key)

# The gke mock plays the google types (crates/dform-mock/schemas/gke.df:
# every one mints its id at Apply). Each deployment configures it from its
# own settings, and refuses to plan when the account the provider
# reports is not the one this env names: the same shell cannot point
# prod's plan at dev's project. Against the real provider the credentials
# would come from the environment, by key, as a secret:
#   credentials = env.var("GOOGLE_CREDENTIALS_{env}")
# and the account from the credentials; the mock reports its `account`.
provider gke {
  project = gcp.project_id
  region = gcp.region
  account = gcp.project_id
  expect_account = gcp.project_id
}

deny "GKE static IPs must be INTERNAL" where gke.static_ips."pngu-grpc".type != "INTERNAL"

# Resources (renfry module intent)

# GKE subnet (terraform-google-modules/network/google//modules/subnets)
resource google.compute_subnetwork gke_subnet {
  project = gcp.project_id
  region = gcp.region
  network = "projects/${gcp.project_id}/global/networks/${gcp.network_name}"
  name = "${gcp.project_name}-gke-subnet"
  ip_cidr_range = gke.subnet_cidr
  subnet_private_access = true
  secondary_ranges = [
    { range_name: "${gcp.project_name}-gke-pods-range", ip_cidr_range: gke.pods_cidr },
    { range_name: "${gcp.project_name}-gke-svc-range", ip_cidr_range: gke.services_cidr }
  ]
}

# Regional managed proxy subnet
resource google.compute_subnetwork regional_proxy {
  project = gcp.project_id
  region = gcp.region
  network = "projects/${gcp.project_id}/global/networks/${gcp.network_name}"
  name = "${gcp.project_name}-proxy"
  purpose = "REGIONAL_MANAGED_PROXY"
  role = "ACTIVE"
  ip_cidr_range = gcp.regional_proxy_cidr
}

# VPC peerings
resource google.compute_network_peering "peering-${name}" {
  project = gcp.project_id
  name
  network = "projects/${gcp.project_id}/global/networks/${gcp.network_name}"
  peer_network = peer
} where peering(env: env, name: name, peer_network: peer)

# Static IPs for services
resource google.compute_address "static-ip-${ip}" {
  project = gcp.project_id
  region = gcp.region
  name = ip
  address_type = gke.static_ips."pngu-grpc".type
  subnetwork_id = gke_subnet
} where ip = "pngu-grpc"

########################################
# Resources (GKE module intent)
########################################

# Represents the external terraform-mi-gke module invocation.
resource google.container_cluster pngu {
  cluster_type = "autopilot-private"
  env
  project_id = gcp.project_id
  project_name = gcp.project_name
  name = "${gcp.project_name}-gke"
  network_id = "projects/${gcp.project_id}/global/networks/${gcp.network_name}"
  subnetwork_id = gke_subnet
  master_control_plane_cidr = gke.control_plane_cidr
  ip_range_pods = "${gcp.project_name}-gke-pods-range"
  ip_range_services = "${gcp.project_name}-gke-svc-range"

  enable_secret_manager_addon = true
  deletion_protection = false
}

########################################
# Resources (monitoring module intent)
########################################

# The dashboard JSON is a file, a document `json(path)` reads.
resource google.monitoring_dashboard pngu {
  project = gcp.project_id
  dashboard_json = json("files/dashboard-pngu.json")
}

# Represents terraform-gcp-alert-metrics module invocation. No provider
# schema has the module, so the program declares its type (the mock plays
# it).
type alert_metrics_pack {
  project_id: string
  notification_channel_type: string
  notification_channel_display_name: string
  enable_standard_gke_alerts: bool
  source: string
}

resource alert_metrics_pack pngu {
  project_id = gcp.project_id
  notification_channel_type = "slack"
  notification_channel_display_name = "Slack"
  enable_standard_gke_alerts = true
  source = "git@github.com:coherentpath/terraform-gcp-alert-metrics.git?ref=v1.5.0"
}
