# pngu.df
#
# dform representation of ~/src/movableink/pngu/terraform.
#
# To run it, from this directory: README.md.
#
# Each env value is its own deployment with its own state:
# dform.state/pngu/env=prod/ never sees dev's resources. Each env is its own
# GCP project (settings below), so names may repeat across envs: isolated.
#
# Notes:
# - This file models the intent/config surface area of the Terraform stack.
# - External Terraform modules are represented as higher-level resources.
# - The monitoring dashboard JSON is read with the file provider from a stub,
#   files/dashboard-pngu.json (see terraform/modules/monitoring/dashboard-pngu.json).

edition 2026
stack pngu[env] { isolated = true }
########################################
# Schema declarations
########################################

# The file provider answers file.json once its path is known.
extern file.json(+path, -value)

########################################
# Environment selection
########################################

# The environments, one type for the key and the peering table's column.
type environment = enum("dev", "stg", "prod")

# A typed stack input: --set env=qa is an error naming the input.
input env: environment = "dev"

########################################

########################################

# The selected environment's settings. Every use reads env and one leaf.
let cfg = settings[env]

########################################
# Provider (configured by the key)
########################################

# The gke mock plays the google types (crates/dform-mock/schemas/gke.df:
# every one mints its id at Apply). Each deployment configures it from its
# own settings row, and refuses to plan when the account the provider
# reports is not the one this env names: the same shell cannot point
# prod's plan at dev's project. Against the real provider the credentials
# would come from the environment, by key, as a secret:
#   credentials = env_var("GOOGLE_CREDENTIALS_{env}")
# and the account from the credentials; the mock reports its `account`.
provider gke {
  project = cfg.project_id
  region = cfg.region
  account = cfg.project_id
  expect_account = cfg.project_id
}

########################################
# Peerings (a table: one row per peering, every env's)
########################################

# The rows are data/peerings.csv's, typed column by column: a row whose env
# is not one of these is an error naming its line.
input relation peering(env: environment, name: string, peer_network: string) from csv("data/peerings.csv")

########################################
# Validation
########################################

constraint "GKE static IPs must be INTERNAL" if cfg.gke.static_ips."pngu-grpc".type != "INTERNAL"

########################################
# Resources (renfry module intent)
########################################

# GKE subnet (terraform-google-modules/network/google//modules/subnets)
resource google_compute_subnetwork gke_subnet {
  project = cfg.project_id
  region = cfg.region
  network = "projects/{cfg.project_id}/global/networks/{cfg.network_name}"
  name = "{cfg.project_name}-gke-subnet"
  ip_cidr_range = cfg.gke.subnet_cidr
  subnet_private_access = true
  secondary_ranges = [
    { range_name: "{cfg.project_name}-gke-pods-range", ip_cidr_range: cfg.gke.pods_cidr },
    { range_name: "{cfg.project_name}-gke-svc-range", ip_cidr_range: cfg.gke.services_cidr }
  ]
}

# Regional managed proxy subnet
resource google_compute_subnetwork regional_proxy {
  project = cfg.project_id
  region = cfg.region
  network = "projects/{cfg.project_id}/global/networks/{cfg.network_name}"
  name = "{cfg.project_name}-proxy"
  purpose = "REGIONAL_MANAGED_PROXY"
  role = "ACTIVE"
  ip_cidr_range = cfg.regional_proxy_cidr
}

# VPC peerings
resource google_compute_network_peering "peering-{name}" {
  for peering{ env: env, name: name, peer_network: peer }
  project = cfg.project_id
  name = name
  network = "projects/{cfg.project_id}/global/networks/{cfg.network_name}"
  peer_network = peer
}

# Static IPs for services
resource google_compute_address "static-ip-{ip}" {
  for ip = "pngu-grpc"
  project = cfg.project_id
  region = cfg.region
  name = ip
  address_type = cfg.gke.static_ips."pngu-grpc".type
  subnetwork_id = gke_subnet.id
}

########################################
# Resources (GKE module intent)
########################################

# Represents the external terraform-mi-gke module invocation.
resource gke_cluster pngu {
  cluster_type = "autopilot-private"
  env = env
  project_id = cfg.project_id
  project_name = cfg.project_name
  name = "{cfg.project_name}-gke"
  network_id = "projects/{cfg.project_id}/global/networks/{cfg.network_name}"
  subnetwork_id = gke_subnet.id
  master_control_plane_cidr = cfg.gke.control_plane_cidr
  ip_range_pods = "{cfg.project_name}-gke-pods-range"
  ip_range_services = "{cfg.project_name}-gke-svc-range"

  enable_secret_manager_addon = true
  deletion_protection = false
}

########################################
# Resources (monitoring module intent)
########################################

# The dashboard JSON is a file, read by the file provider.
resource google_monitoring_dashboard pngu {
  project = cfg.project_id
  dashboard_json = file.json["files/dashboard-pngu.json"]
}

# Represents terraform-gcp-alert-metrics module invocation.
resource alert_metrics_pack pngu {
  project_id = cfg.project_id
  notification_channel_type = "slack"
  notification_channel_display_name = "Slack"
  enable_standard_gke_alerts = true
  source = "git@github.com:coherentpath/terraform-gcp-alert-metrics.git?ref=v1.5.0"
}

########################################
# Optional outputs (handy for wiring)
########################################

output active_env = env
output project_id = cfg.project_id
