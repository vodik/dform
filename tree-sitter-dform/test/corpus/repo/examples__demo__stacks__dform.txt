# dform demo: component-scoped resources + settings tables + IAM policy assembly.
#
# To run it, from this directory: README.md.

edition 2026

import "modules/network.df"
import "modules/database.df"
import "modules/kubernetes.df"
import "modules/iam.df"
import "policies/baseline.df"
import "modules/stdlib_net.df"

# The stack is this file, dform, keyed by env: dform[env=prod]'s state is dform.state/dform/env=prod/.
#| The deployment's environment, the stack's key: `dform plan dform env=prod`
#| names the deployment, and without one the default is the deployment.
#| owner: platform
key env: environment = "staging"

data("zone", "us-test-1a")
data("zone", "us-test-1b")

# A fact with a condition: it holds in prod only.
iam_need("app", "audit.write", "audit-log") where env == "prod"
# A module's need is its output, wired here.
iam_need("app", n.action, n.resource) where n = database.main.iam_need
iam_need("app", n.action, n.resource) where n = kubernetes.main.iam_need

# Settings (branch as data). One @default layer for every settings row that
# exists; each environment's config/dform/{env}.yaml (the stack's config) wins
# per leaf at the normal rank. A CIDR is a string there, made an inet where
# it is used.
settings _ @default {
  network.main.vpc_net = "10.50.0.0/16"
  network.peer.vpc_net = "10.60.0.0/16"
  db = { backup_days: 3, multi_az: false }
  k8s = { private_api: false, nodepool: { min: 1, max: 3 } }
}

# The settings of the selected environment, by name: every use reads env
# and one leaf.
let cfg = settings[env]

# The provider (the mock's built-in fake schema). The stack's settings are dform.toml's [stacks.dform]:
# strict, isolated (each env in its own account, so identity's fixed names do not collide) and config
# (each env's settings are config/dform/{env}.yaml's, a leaf per contribution).
provider fake {}
#| The environments: an alias, the enum wherever it is written.
type environment = enum("dev", "staging", "prod")

use baseline
use stdlib_net

# Instantiate modules (Terraform module-like). Inputs are declared by the
# module; each `k = v` here is a contribution to one.
instance network main {
  vpc_net = inet(cfg.network.main.vpc_net)
}

# Dev is small: one network, no database, no cluster.
instance network peer {
  vpc_net = inet(cfg.network.peer.vpc_net)
} where env != "dev"

# Cross-instance values are wired through outputs.
instance database main {
  backup_days = cfg.db.backup_days
  multi_az = cfg.db.multi_az
  subnet_ids = network.main.private_subnet_ids
} where env != "dev"

instance kubernetes main {
  private_api = cfg.k8s.private_api
  nodepool_min = cfg.k8s.nodepool.min
  nodepool_max = cfg.k8s.nodepool.max
  subnet_ids = network.main.private_subnet_ids
} where env != "dev"

instance identity main {}

#| VPC peering intent: define edges between network instances.
vpc_peer_inst("main", "peer")

#| Each network instance outputs its vpc's address; an edge's pair is the
#| two instances' vpcs, joined on the instance names.
vpc_peer_pair(ia, ib, a, b) where vpc_peer_inst(ia, ib), a = network[ia].vpc, b = network[ib].vpc
vpc_peer(a, b) where vpc_peer_pair(_, _, a, b)

#| A peering resource per declared peering edge.
resource net.vpc_peering "peer-${ia}-${ib}" {
  # Both ends must exist in this environment.
  requester_vpc_id = a.id
  accepter_vpc_id = b.id
} where vpc_peer_pair(ia, ib, a, b), a in net.vpc, b in net.vpc

#| Demo: allocate an IP from the main VPC.
resource compute.vm bastion {
  private_ip = inet.host(inet(cfg.network.main.vpc_net), 20)
}

# Policy, over every resource of a type (a module's resources included).
deny "prod db must be multi_az" where env == "prod", pg in db.postgres, not pg.multi_az

deny "prod k8s must have private api" where env == "prod", c in k8s.cluster, not c.private_api

#| Policy over hypothetical inputs: `dform test` runs every scenario
#| against an empty mock world; `dform plan --scenario prod` plans the stack
#| as if env were prod.
scenario prod {
  set env = "prod"
  deny "prod peers the two vpcs" where not "peer-main-peer" in net.vpc_peering
  deny "prod keeps 14 days of db backups" where not db.postgres["database.main::db"].backup_days == 14
}

scenario dev {
  set env = "dev"
  deny "dev has no database" where _ in db.postgres
  deny "dev has no peer network" where "network.peer::vpc" in net.vpc
}
