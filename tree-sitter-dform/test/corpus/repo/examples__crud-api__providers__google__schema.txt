# A mock of the Google provider's types crud_api.df uses: a VPC and a Cloud
# SQL instance, database and user. The instance's private IP is computed
# (an open null until it exists); the user's password is sensitive.

edition 2026

type_provider("google_compute_network", "google")
type_attr("google_compute_network", "id", "string", ["computed", "id"])
type_attr("google_compute_network", "name", "string", ["required", "force_new"])
type_attr("google_compute_network", "auto_create_subnetworks", "bool", [])

type_provider("google_sql_database_instance", "google")
type_attr("google_sql_database_instance", "id", "string", ["computed", "id"])
type_attr("google_sql_database_instance", "name", "string", ["required", "force_new"])
type_attr("google_sql_database_instance", "region", "string", ["required", "force_new"])
type_attr("google_sql_database_instance", "database_version", "string", ["required"])
type_attr("google_sql_database_instance", "deletion_protection", "bool", [])
type_attr("google_sql_database_instance", "settings", "object", [])
type_attr("google_sql_database_instance", "settings.tier", "string", ["required"])
type_attr("google_sql_database_instance", "settings.availability_type", "string", [])
type_attr("google_sql_database_instance", "settings.backup_configuration", "object", [])
type_attr("google_sql_database_instance", "settings.backup_configuration.enabled", "bool", [])
type_attr("google_sql_database_instance", "settings.backup_configuration.point_in_time_recovery_enabled", "bool", [])
type_attr("google_sql_database_instance", "settings.ip_configuration", "object", [])
type_attr("google_sql_database_instance", "settings.ip_configuration.ipv4_enabled", "bool", [])
type_attr("google_sql_database_instance", "settings.ip_configuration.private_network", "ref(google_compute_network)", [])
type_attr("google_sql_database_instance", "private_ip_address", "string", ["computed"])
type_mint("google_sql_database_instance", "private_ip_address", "10.20.0.3")

type_provider("google_sql_database", "google")
type_attr("google_sql_database", "id", "string", ["computed", "id"])
type_attr("google_sql_database", "instance", "string", ["required", "force_new"])
type_attr("google_sql_database", "name", "string", ["required", "force_new"])

type_provider("google_sql_user", "google")
type_attr("google_sql_user", "id", "string", ["computed", "id"])
type_attr("google_sql_user", "instance", "string", ["required", "force_new"])
type_attr("google_sql_user", "name", "string", ["required", "force_new"])
type_attr("google_sql_user", "password", "string", ["sensitive"])
