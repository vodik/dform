# adv7b: A reads B.x to write A.y; B reads A.z to write B.x. Not a cycle
# under (type, path) partitions, PROVIDED the reads are on separate rules.
# (The resource-block sugar shares one body across every field, so putting
# the read in the block would drag A.z above B.x and close the cycle.)

edition 2026

  z = 1
}

  want(net.vpc, "x"),
  arg(net.subnet, "y", .x, p)

resource net.subnet y {
  for arg(net.vpc, "x", .z, q)
  x = q
}
