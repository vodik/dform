# The baseline every stack applies, `use baseline`: a team tag, audit
# sinks, no public database. `env` is the using stack's; its input is the
# stack's to give, by its path (config/dform/prod.yaml's `baseline.audit`).
input audit { enabled: bool = false, sinks: list(string) = ["s3"] }
# Lattices for paths several authors contribute to. Object paths such as
# `tags` are maps by default; a list is one value unless declared a set.
type_lattice(iam.policy, "statements", "set")

# Example: add a shared tag to everything.
set r.tags = { team: "platform" } where r in resource

# Example: policy pack can contribute directly to an IAM policy. Reads
# .name of the type it writes .statements on: legal under (type, path)
# partitions.
set p.statements = [{ action: "org.read", resource: "org" }] where {
  env == "prod"
  p in iam.policy
  p.name == "app"
}

deny "db must not be public" { resource: pg } where pg in db.postgres, not pg.public == false

#| Prod should enable audit logging.
warn "prod should enable audit logging" { env: "prod" } where env == "prod", not audit.enabled

# Networking invariants live in `stdlib_net.df`.
