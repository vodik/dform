# The workload stack of the bootstrap demo (README "Controller mode"): a
# namespace, a deployment and a service on the mock Kubernetes provider,
# whose image comes from an input relation. After `dform stack handover`
# the dform-controller Deployment that bootstrap.df installs runs it.
# To run it, from this directory: README.md.
#
# Edit data/release.facts and the controller deploys it; edit the world and
# the replicas drift is corrected, the image drift is held for approval.
# In prod a rollout waits for a signed approval of the plan's digest
# (README "Approvals"): the controller publishes the digest in
# approval-pending.json beside the state, and a token for it, from the
# example signer, releases it:
#
#   cargo run -q -p dform-direct --bin dform-approve -- sign approver.key \
#     --digest sha256:... --stack renfry.workload --approver alice --format fact >> data/approvals.facts

edition 2026

stack renfry.workload {
  # Whose signatures approve a plan: `dform-approve keygen approver.key`
  # prints this JWKS.
  approvals = jwks_file("approvers.jwks.json")
}

provider k8s {}

# The release to run, `release(image)`; re-read whenever the file changes.
input release(image) from facts("data/release.facts")
# `approve(t, a)`: accept the drift of t.a, correcting it.
input approve(type, address) from facts("data/approvals.facts")
# `approval(token)`: a signed approval of a plan's digest.
input approval(token) from facts("data/approvals.facts")

input env: enum("dev", "prod") = "dev"

resource k8s.namespace web {
  metadata.name = "web"
}

resource k8s.deployment web {
  if release(image)
  metadata.name = "web"
  metadata.namespace = k8s.namespace["web"].metadata.name
  spec.replicas = 3
  spec.selector.matchLabels = { app: "web" }
  spec.template.metadata.labels = { app: "web" }
  spec.template.spec.containers = [{
    name: "web",
    image,
    ports: [{ containerPort: 8080, protocol: "TCP" }]
  }]
}

resource k8s.service web {
  metadata.name = "web"
  metadata.namespace = k8s.namespace["web"].metadata.name
  spec.selector = { app: "web" }
  spec.ports = [{ port: 80, targetPort: 8080, protocol: "TCP" }]
}

# Replicas are the autoscaler's business on the way back: a drift there is
# corrected without asking. Any other drift waits for `approve`.
auto_reconcile(t, a, p) if drift(t, a, p, _, _), p == "spec.replicas"

# In prod a rollout needs an approval: the controller holds it until a
# token for the plan's digest arrives.
requires_approval(d, "a prod rollout") if {
  env == "prod"
  deformation(k, k8s.deployment, a, _)
  k != "delete"
  d = "k8s.deployment[\"${a}\"]"
}
