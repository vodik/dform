edition 2026

input backup_days: int
input multi_az: bool
#| The subnets the database is placed in, one row each.
input subnet

decl subnet(subnet: net.subnet)

#| A PostgreSQL database in the network's private subnets.
resource db.postgres db {
  public = false
  subnets = [ s | subnet(s), s in net.subnet ]
  backup_days
  multi_az
}

#| Only when the database exists in this environment.
output iam_need = { action: "db.connect", resource: db } where has backup_days
