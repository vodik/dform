module network {
  input vpc_net: inet

  resource net.vpc vpc {
    cidr = vpc_net
    tags = { env, component: "network" }
  }

  zone_index("us-test-1a", 0)
  zone_index("us-test-1b", 1)

  resource net.subnet "private-${z}" {
    vpc_id = vpc.id
    cidr = inet.subnet(vpc.cidr, 4, zone_index[z])
    zone = z
    visibility = "private"
  } where data("zone", z)

  output vpc: net.vpc = vpc
  output private_subnet_ids: list(ref(net.subnet)) = [ s.id | s in net.subnet, s.visibility == "private" ]
}
