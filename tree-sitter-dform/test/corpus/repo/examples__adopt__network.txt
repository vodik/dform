=========================
examples/adopt/network.df
=========================

edition 2026

input vpc_net: inet

resource net.vpc vpc {
  cidr = vpc_net
  tags = { env, component: "network" }
}

zone_index("us-test-1a", 0)
zone_index("us-test-1b", 1)

resource net.subnet "private-${z}" {
  vpc
  cidr = inet.subnet(vpc.cidr, 4, zone_index[z])
  zone = z
  visibility = "private"
} where data("zone", z)

output vpc: net.vpc = vpc
output private_subnets: list(ref(net.subnet)) = [ s | s in net.subnet, s.visibility == "private" ]
