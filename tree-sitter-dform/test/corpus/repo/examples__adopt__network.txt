input vpc_net: inet
#| The private subnets, one row each.
decl private_subnet(subnet: net.subnet)
output vpc: net.vpc = vpc
output private_subnet

resource net.vpc vpc { cidr = vpc_net, tags = { env, component: "network" } }

zone_index("us-test-1a", 0)
zone_index("us-test-1b", 1)

resource net.subnet "private-${z}" {
  vpc
  cidr = inet.subnet(vpc.cidr, 4, zone_index[z])
  zone = z
  visibility = "private"
} where data("zone", z)

private_subnet(s) where s in net.subnet, s.visibility == "private"
