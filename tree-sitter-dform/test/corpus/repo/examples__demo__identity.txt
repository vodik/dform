# A component with no inputs: its file declares resources. `iam_need` is
# the stack's relation, read outward.

#| The app's role, and one policy of every permission the stack's other
#| components need.
resource iam.role app_role { name = "app", assume = { principals: ["k8s.serviceaccount.app"] } }

resource iam.policy app_policy {
  name = "app"
  # List comprehension sugar for collect_list(...)
  statements = [ { action: a, resource: r } | iam_need("app", a, r) ]
}

resource iam.role_policy_attachment attach { role = app_role, policy = app_policy }
