# Demo: adopting an existing resource from inventory.
#
# Staging creates the VPC; prod adopts the one the inventory already has.
# To run it, from this directory: README.md.

edition 2026
stack adopt_demo {}
input env: enum("staging", "prod") = "staging"

settings staging {
  network.vpc_cidr = inet("10.10.0.0/16")
}

settings prod {
  network.vpc_cidr = inet("10.20.0.0/16")
}

import "modules/network.df"

instance network main {
  vpc_net = settings[env].network.vpc_cidr
}

# Adopt the VPC from inventory, but manage it under the local name `network.main::vpc`.
adopt(net.vpc, network.main/vpc, "existing-prod-vpc") if
  env == "prod",
  "existing-prod-vpc" in world.net.vpc

# Example: use discovered inventory values directly in config.
network.main/vpc.adopted_id = cloud_ref(net.vpc, "existing-prod-vpc", .id) if env == "prod"

# Keep the existing owner tag stable (avoids a "remove tag" diff on adopt).
network.main/vpc.tags.owner = cloud_ref(net.vpc, "existing-prod-vpc", .tags.owner) if env == "prod"

# Also copy it into a separate attribute to demonstrate dotted path lookup.
network.main/vpc.owner_tag = cloud_ref(net.vpc, "existing-prod-vpc", .tags.owner) if env == "prod"
