# Demo: adopting an existing resource from inventory.
#
# Staging creates the VPC; prod adopts the one the inventory already has.
# To run it, from this directory: README.md.

edition 2026

input env: enum("staging", "prod") = "staging"

# The mock's built-in fake schema.
provider fake

# network.df's input, per environment.
set { network.vpc_net = inet("10.10.0.0/16") } where env == "staging"
set { network.vpc_net = inet("10.20.0.0/16") } where env == "prod"

# network.df, imported once, its input the settings': its VPC is
# `network::vpc`.
use network

# Adopt the VPC from inventory, but manage it under the local name `network::vpc`.
adopt(net.vpc["network::vpc"], "existing-prod-vpc") where {
  env == "prod"
  "existing-prod-vpc" in world.net.vpc
}

# Example: use discovered inventory values directly in config.
set net.vpc["network::vpc"].adopted_id = cloud_ref(net.vpc, "existing-prod-vpc", "id") where env == "prod"

# Keep the existing owner tag stable (avoids a "remove tag" diff on adopt).
set net.vpc["network::vpc"].tags.owner = cloud_ref(net.vpc, "existing-prod-vpc", "tags.owner") where env == "prod"

# Also copy it into a separate attribute to demonstrate dotted path lookup.
set net.vpc["network::vpc"].owner_tag = cloud_ref(net.vpc, "existing-prod-vpc", "tags.owner") where env == "prod"
