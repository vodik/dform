# dform-advanced.df
#
# Goal: show fixpoint-style inference (transitive closure) applied to real infra problems:
# - derive reachability between VPCs from peering edges
# - auto-generate route resources for every reachable pair
# - enforce policies based on derived reachability (tenant isolation)
# - demonstrate another closure: nested group membership
#
# To run it, from this directory: README.md.

input env: enum("staging", "prod") = "staging"
# List-valued inputs (human-friendly), the same in every environment.
input tenants: list(string) = ["tenant_a", "tenant_b"]
input mesh {
  vpcs: list(any) = [
    { name: "core_a", net: inet("10.210.0.0/16") },
    { name: "core_b", net: inet("10.220.0.0/16") },
    { name: "core_c", net: inet("10.230.0.0/16") },
  ]
}
# Peering intent.
# In staging, only tenant_a peers with shared.
# In prod, tenant_b also peers with shared, which creates tenant->tenant reachability via shared.
# A fact and a guarded fact (a rule): declared mixed (E 2.6).
decl peer_intent(tenant) mixed
# Declared so the check can negate it with no rows; uncomment the fact to
# explicitly allow certain non-adjacent direct routes:
decl mesh_allow_direct_route(from, to)

# The mock's built-in fake schema.
provider fake

# Base IP space for tenants
corp_net(inet("10.100.0.0/12"))

# Explode tenant list into a relation (Pattern A), with index.
tenant(t, i) where t = tenants[i]

# Shared services VPC (always exists)
resource net.vpc shared { cidr = inet("10.200.0.0/16"), tags = { segment: "shared" } }

# One VPC per tenant, with deterministic CIDRs derived from corp_net.
resource net.vpc t {
  cidr = inet.subnet(corp, 4, i)
  tags = { segment: "tenant", tenant: t }
} where tenant(t, i), corp_net(corp)

peer_intent("tenant_a")
peer_intent("tenant_b") where env == "prod"

peer("shared", t) where peer_intent(t), tenant(t, _)

# Convert peer intent into directed graph edges.
# (Peering is bidirectional in this demo.)
link(a, b) where peer(a, b)
link(b, a) where peer(a, b)

# Fixpoint 1: transitive closure for reachability.
reachable(a, b) where link(a, b)
reachable(a, c) where reachable(a, b), link(b, c)

# Derived infra: create a route for every reachable pair (excluding self).
# `dest` is bound in the clause, so it is read now (an attr join);
# `destination = net.vpc[b].cidr` would be a reference.
resource net.route "rt-${a}-${b}" {
  from = a
  to = b
  destination = dest
} where reachable(a, b), a != b, dest = net.vpc[b].cidr

# Policy: warn if we materialize *direct* routes between non-adjacent nodes in the mesh.
#
# This highlights an important distinction:
# - `reachable/2` is the *closure* (it includes transitive reachability)
# - a `net.route` is a *direct* configuration object
#
# In some orgs you may want transitive reachability without generating a direct
# route for every transitive pair (or you may want to require explicit allows).
mesh_node(n) where mesh_vpc(n, _, _)

mesh_adjacent(a, b) where mesh_peer(a, b)
mesh_adjacent(b, a) where mesh_peer(a, b)

mesh_non_adjacent_pair(a, b) where {
  mesh_node(a)
  mesh_node(b)
  a != b
  reachable(a, b)
  not mesh_adjacent(a, b)
}

warn "mesh has direct route between non-adjacent nodes" { from: a, to: b } where {
  mesh_non_adjacent_pair(a, b)
  "rt-${a}-${b}" in net.route
  not mesh_allow_direct_route(a, b)
}

# mesh_allow_direct_route("core_a", "core_c")

# --- Auto-mesh demo (derived links from a list) ---
#
# The mesh itself is described as a *list setting* for human ergonomics.
# We immediately explode it into row facts, then derive edges, then rely on
# reachability closure to fill in the implied connectivity.

mesh_vpc(name, cidr, i) where mesh.vpcs[i] = { name, net: cidr }

resource net.vpc name { cidr, tags = { segment: "mesh" } } where mesh_vpc(name, cidr, _)

# Derive a sparse base graph: connect consecutive VPCs in the list.
# The reachability closure will infer core_a -> core_c via core_b.
mesh_peer(a, b) where {
  mesh_vpc(a, cidr_a, ia)
  mesh_vpc(b, cidr_b, ib)
  ib == ia + 1
  not inet.overlaps(cidr_a, cidr_b)
}

# Feed the derived edges into the global peering/link graph.
peer(a, b) where mesh_peer(a, b)

# Optional: materialize the derived peerings as explicit peering resources.
resource net.vpc_peering "mesh-peer-${a}-${b}" {
  requester_vpc = net.vpc[a]
  accepter_vpc = net.vpc[b]
} where mesh_peer(a, b)

# Policy 1: peered VPCs must not overlap.
deny "peered vpcs must not overlap" { a, b } where {
  peer(a, b)
  inet.overlaps(net.vpc[a].cidr, net.vpc[b].cidr)
}

# Policy 2: tenant isolation.
# Only enforced in prod, and depends on the reachability closure.
deny "tenant isolation violated" { a, b } where {
  env == "prod"
  tenant(a, _)
  tenant(b, _)
  a != b
  reachable(a, b)
}

# Fixpoint 2: nested group membership.
# This is another common closure problem in real org infra.
group_edge("sre", "admins")
group_edge("oncall", "sre")

group_member("alice", "oncall")
group_member("bob", "sre")

in_group(u, g) where group_member(u, g)
in_group(u, g2) where in_group(u, g1), group_edge(g1, g2)

warn "user is effectively admin (via group nesting)" { user: u } where in_group(u, "admins")
