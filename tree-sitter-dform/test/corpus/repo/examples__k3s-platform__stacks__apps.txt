#| What runs on the cluster: the web app, its Service and its name.
key env: enum("staging", "prod") = "staging"
input replicas: int = 2

use stacks.platform
use ovh { project = "shop" }
use k8s { kubeconfig = platform[env].kubeconfig }
use policy

set replicas = 3 where env == "prod"

resource k8s.deployment web {
  metadata = { name: "web", namespace: "apps" }
  spec.replicas = replicas
  spec.selector.matchLabels = { app: "web" }
  spec.template.metadata.labels = { app: "web" }
  spec.template.spec.containers = [{
    name: "web",
    image: "nginx:1.27@sha256:c4ab9b7d9cdbba02d12e91035d6350386d1c05daa28b1f5c816f3d3f34c75147",
    ports: [{ containerPort: 80, protocol: "TCP" }],
  }]
}

resource k8s.service web {
  metadata = { name: "web", namespace: "apps" }
  spec.selector = { app: "web" }
  spec.ports = [{ port: 80, targetPort: 80, protocol: "TCP" }]
}

#| The site's name, the cluster's address.
resource ovh.domain_record www {
  zone = "example.com"
  subdomain = "www.${env}"
  type = "A"
  target = platform[env].ingress
  ttl = 300
}
