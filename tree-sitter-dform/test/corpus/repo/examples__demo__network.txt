# The demo's networking: a component, `vpc`, copied once per network
# (`instance network.vpc main`). A copy's private subnets are a relation
# it exports, `output private_subnet`: a database or a cluster takes them
# a row at a time, `subnet(s) where main.private_subnet(s)`, where a list
# would be handed over whole (docs/grammar.md "Inputs and outputs": a list
# is a value handled whole, anything iterated or joined is a relation).

#| One VPC, and a private subnet in every zone.
component vpc {
  #| The VPC's IPv4 range; each zone's subnet is a /20 of it.
  input vpc_net: inet
  decl private_subnet(subnet: net.subnet)
  #| The network's VPC, for peering.
  output vpc: net.vpc = vpc
  #| The network's private subnets, one row each, for a database or a
  #| cluster: `main.private_subnet(s)`.
  output private_subnet

  resource net.vpc vpc { cidr = vpc_net, tags = { env, component: "network" } }

  zone_index("us-test-1a", 0)
  zone_index("us-test-1b", 1)

  resource net.subnet "private-${z}" {
    vpc
    cidr = inet.subnet(vpc.cidr, 4, zone_index[z])
    zone = z
    visibility = "private"
  } where data("zone", z)

  private_subnet(s) where s in net.subnet, s.visibility == "private"
}
