# A small web stack planned against the Terraform-shaped mock AWS provider
# (dform.toml: aws is the built-in aws-mock). No account: the world is a JSON file.
#
# To run it, from this directory: README.md.
#
# Two shapes to look at:
#   - Optional+Computed: subnet `a` sets availability_zone, so a ref to it is
#     the constant; instance `web` leaves its own unset, so a ref to it is the
#     null ?aws_instance/web#availability_zone until Apply picks one. Instance
#     `bastion` leaves subnet_id unset: AWS puts it in a default subnet.
#   - keyless sets: the security group's ingress rules have no key; the plan
#     compares them as a set, so their order never matters.

edition 2026
provider aws {}

resource aws_vpc main {
  cidr_block = "10.0.0.0/16"
  tags = { "Name": "main" }
}

resource aws_subnet a {
  vpc_id = main.id
  cidr_block = "10.0.1.0/24"
  availability_zone = "us-east-1a"
}

resource aws_security_group web {
  name = "web"
  vpc_id = main.id
  ingress = [
    { from_port: 443, to_port: 443, protocol: "tcp", cidr_blocks: ["0.0.0.0/0"] },
    { from_port: 80, to_port: 80, protocol: "tcp", cidr_blocks: ["0.0.0.0/0"] }
  ]
  egress = [
    { from_port: 0, to_port: 0, protocol: "-1", cidr_blocks: ["0.0.0.0/0"] }
  ]
}

resource aws_instance web {
  ami = "ami-0abcdef1234567890"
  instance_type = "t3.small"
  subnet_id = a.id
  vpc_security_group_ids = [aws_security_group["web"].id]
}

resource aws_instance bastion {
  instance_type = "t3.micro"
  tags = { subnet_az: a.availability_zone, web_az: aws_instance["web"].availability_zone }
}

resource aws_eip web {
  domain = "vpc"
  instance = aws_instance["web"].id
}

resource aws_db_instance app {
  identifier = "app"
  instance_class = "db.t3.micro"
  password = "correct-horse-battery-staple"
  vpc_security_group_ids = [aws_security_group["web"].id]
  skip_final_snapshot = true
}
