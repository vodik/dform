input private_api: bool
input nodepool_min: int
input nodepool_max: int
#| The subnets the cluster is placed in, one row each.
input subnet
decl subnet(subnet: net.subnet)
#| Only when the cluster exists in this environment.
output iam_need = { action: "k8s.describe_cluster", resource: cluster } where has private_api

#| A Kubernetes cluster and one node pool in the network's private
#| subnets.
resource k8s.cluster cluster {
  version = "1.29"
  private_api
  subnets = [ s | subnet(s), s in net.subnet ]
}

resource k8s.nodepool default { cluster, min = nodepool_min, max = nodepool_max }
