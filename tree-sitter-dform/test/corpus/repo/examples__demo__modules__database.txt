# The network module's `subnets` type.
import "modules/network.df"
#| A PostgreSQL database in the network's private subnets.
module database {
  input backup_days: int
  input multi_az: bool
  input subnet_ids: subnets
  contributes iam_need

  resource db.postgres db {
    public = false
    subnet_ids = subnet_ids
    backup_days = backup_days
    multi_az = multi_az
  }

  #| Only when the database exists in this environment.
  iam_need("app", "db.connect", db.id) where has backup_days
}
