examples/gke/stacks/gke_one_zone.df

# gke_two_phase.df with a cluster in one zone: E §2.7 worked example item 6,
# the other branch. Tick 1 creates the cluster; its zones come back with one
# element, so at the boundary deny(two_zones) derives and apply stops before
# touching the nodepools and the kubernetes objects:
#   cargo run -- -C examples/gke apply gke_one_zone
#
# The rest is C's two-phase GKE / kubernetes stack (E §7.4), in the current syntax.
# Run with the gke mock schema, which makes the cluster's zones, endpoint and
# ca open nulls until the cluster exists:
#   cargo run -- -C examples/gke plan gke_two_phase

edition 2026
stack gke_one_zone {}
provider gke {}

input env: enum("dev", "stg", "prod") = "dev"

settings dev {
  project_id = "renfry-dev-973682"
  project_name = "renfry-dev"
  region = "us-east1"
  network_name = "renfry-dev-network"
  gke = {
    control_plane_cidr: inet("172.16.3.96/28"),
    subnet_cidr: inet("10.141.76.0/22")
  }
}

let cfg = settings[env]

project(cfg.project_id, cfg.project_name, cfg.region,
  "projects/{cfg.project_id}/global/networks/{cfg.network_name}")

resource google_compute_subnetwork gke_subnet {
  for project(id, name, region, net_id)
  project = id
  region = region
  network = net_id
  name = "{name}-gke-subnet"
  ip_cidr_range = cfg.gke.subnet_cidr
}

# Carries a fresh null (subnetwork_id) and is still definite.
resource google_compute_address static_ip {
  for project(id, _, region, _)
  project = id
  region = region
  name = "pngu-grpc"
  subnetwork_id = gke_subnet.id
}

resource gke_cluster pngu {
  for project(id, name, _, net_id)
  env = env
  project_id = id
  name = "{name}-gke"
  network_id = net_id
  subnetwork_id = gke_subnet.id
  master_control_plane_cidr = cfg.gke.control_plane_cidr
  node_locations = ["us-east1-b"]
  deletion_protection = true
}

# Provider configured from phase-1 nulls (two open, one secret).
provider_config("kubernetes", {
  host: gke_cluster.pngu.endpoint,
  ca: gke_cluster.pngu.ca_certificate,
  token: google.client_config["current"].access_token
})

resource k8s.namespace pngu {
  name = "pngu"
}

# A read in the clause is an attr join (the value now), not a reference.
resource k8s.deployment api {
  if ns = k8s.namespace.pngu.name
  namespace = ns
  image = "gcr.io/renfry/api:1.42"
  replicas = 3
}

resource k8s.secret db_credentials {
  if ns = k8s.namespace.pngu.name
  namespace = ns
  data = { password: google.secret_manager_secret_version["db_pw"].secret_data }
}

# One nodepool per zone: member over an open null is a content position.
resource gke_nodepool "np-{z}" {
  for z in gke_cluster.pngu.zones
  cluster = gke_cluster.pngu.id
  zone = z
  min = 1
  max = 3
}

zone_count(count(z)) if z in gke_cluster.pngu.zones

deny "cluster must be in at least two zones" { cluster: "pngu" } if {
  exists gke_cluster.pngu
  zone_count(n)
  n < 2
}

# A policy that has nothing to do with zones. Under coarse Rule 3 it would be
# undetermined anyway (a `want` rule is stuck), which is the spurious case;
# per key it is decided now, and holds.
deny "cluster must not be deletable" { cluster: c } if c in gke_cluster, not c.deletion_protection
