# README "Approvals": policy says which deformations need an approval, and
# apply verifies a signed approval of the plan's digest before its first
# Apply call. Staging needs none (README.md); a replace in prod needs one:
#
#   cargo run -q -p dform-direct --bin dform-approve -- keygen approver.key > approvers.jwks.json
#   dform apply approvals env=prod
#   dform plan approvals env=prod --set cidr=10.1.0.0/16 --out plan.json
#   cargo run -q -p dform-direct --bin dform-approve -- sign approver.key \
#     --digest sha256:... --stack approvals --key env=prod --approver alice > approval.json
#   dform apply plan.json --approval approval.json

edition 2026

key env: enum("staging", "prod") = "staging"
input cidr: string = "10.0.0.0/16"

# The mock's built-in fake schema.
provider fake

resource net.vpc main {
  cidr = cidr
}

# In prod, replacing anything needs an approval.
requires_approval(d, "a replace in prod") where {
  env == "prod"
  deformation("replace", t, a, _)
  d = "${t}[\"${a}\"]"
}

# Optional: who may approve what. Without it, any key in the trust root may.
approver_allowed(who, d) where requires_approval(d, _), who in ["alice", "bob"]
