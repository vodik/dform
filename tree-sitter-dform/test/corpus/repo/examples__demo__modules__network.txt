#| One VPC, and a private subnet in every zone.
module network {
  #| A type alias, exported: every file that imports this one may write
  #| `subnets` for the type (modules/database.df, modules/kubernetes.df).
  type subnets = list(ref(net.subnet))
  export type subnets
  #| The VPC's IPv4 range; each zone's subnet is a /20 of it.
  input vpc_net: inet

  resource net.vpc vpc {
    cidr = vpc_net
    tags = { env, component: "network" }
  }

  zone_index("us-test-1a", 0)
  zone_index("us-test-1b", 1)

  resource net.subnet "private-${z}" {
    vpc_id = vpc.id
    cidr = inet_subnet(vpc.cidr, 4, zone_index[z])
    zone = z
    visibility = "private"
  } where data("zone", z)
  #| The network's VPC, for peering.
  output vpc: net.vpc = vpc
  output private_subnet_ids: subnets = [ s.id | s in net.subnet, s.visibility == "private" ]
}
