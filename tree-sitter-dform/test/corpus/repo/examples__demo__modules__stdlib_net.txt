# Minimal networking stdlib demo rules/policies.

edition 2026
#| Networking invariants: peered VPCs do not overlap, a VM has an address.
policy stdlib_net {
  #| Deny if peered VPC CIDRs overlap.
  #|
  #| This is intentionally scoped to only VPCs connected by `vpc_peer/2`.
  deny "peered vpc cidr overlaps" { a, b } where {
    vpc_peer(a, b)
    a in net.vpc
    b in net.vpc
    inet.overlaps(a.cidr, b.cidr)
  }

  #| Deny if any VM gets the unspecified IP.
  deny "vm has unspecified ip" { vm } where vm in compute.vm, ip.unspecified(vm.private_ip)
}
