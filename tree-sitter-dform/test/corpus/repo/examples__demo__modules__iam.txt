#| The app's role, and one policy of every permission the modules need.
module identity {
  resource iam.role app_role {
    name = "app"
    assume = { principals: ["k8s.serviceaccount.app"] }
  }

  resource iam.policy app_policy {
    name = "app"
    # List comprehension sugar for collect_list(...)
    statements = [ { action: a, resource: r } | iam_need("app", a, r) ]
  }

  resource iam.role_policy_attachment attach {
    role_id = app_role.id
    policy_id = app_policy.id
  }
}
