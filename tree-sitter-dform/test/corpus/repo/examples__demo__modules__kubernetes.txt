# The network module's `subnets` type.
import "modules/network.df"
#| A Kubernetes cluster and one node pool in the network's private subnets.
module kubernetes {
  input private_api: bool
  input nodepool_min: int
  input nodepool_max: int
  input subnet_ids: subnets
  contributes iam_need

  resource k8s.cluster cluster {
    version = "1.29"
    private_api = private_api
    subnet_ids = subnet_ids
  }

  resource k8s.nodepool default {
    cluster_id = cluster.id
    min = nodepool_min
    max = nodepool_max
  }

  #| Only when the cluster exists in this environment.
  iam_need("app", "k8s.describe_cluster", cluster.id) where has private_api
}
