# adv3: a policy pack that reads attr of the type it contributes to, via a
# DIFFERENT path. E §4.3 says this compiles under (type, path) partitions.
# Reads (attr, iam.policy, name); writes (arg, iam.policy, statements).

edition 2026

resource iam.policy app_policy {
  name = "app"
  statements = [{ action: "db.connect", resource: "db" }]
}

policy p {
  contributes iam.policy.statements
  type_lattice("iam.policy", .statements, "set")
  arg("iam.policy", p, .statements, [{ action: "org.read", resource: "org" }]) if
    want("iam.policy", p),
    arg("iam.policy", p, .name, "app")
}
apply p
