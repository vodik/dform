  peer_intent(tenant_b).

# Convert peer intent into directed graph edges.
# (Peering is bidirectional in this demo.)
link(A, B) :- peer(A, B).
link(B, A) :- peer(A, B).

# Fixpoint 1: transitive closure for reachability.
reachable(A, B) :- link(A, B).
reachable(A, C) :- reachable(A, B), link(B, C).

# Derived infra: create a route for every reachable pair (excluding self).
resource net.route R {
  from = A
  to = B
  destination = NetB
} :-
  reachable(A, B),
  A != B,
  arg(net.vpc, B, .cidr, NetB),
  R = format("rt-%s-%s", A, B).

# Policy: warn if we materialize *direct* routes between non-adjacent nodes in the mesh.
#
# This highlights an important distinction:
# - `reachable/2` is the *closure* (it includes transitive reachability)
# - a `net.route` is a *direct* configuration object
#
# In some orgs you may want transitive reachability without generating a direct
# route for every transitive pair (or you may want to require explicit allows).
mesh_node(N) :- mesh_vpc(N, _Net, _I).

mesh_adjacent(A, B) :- mesh_peer(A, B).
mesh_adjacent(B, A) :- mesh_peer(A, B).

mesh_non_adjacent_pair(A, B) :-
  mesh_node(A),
  mesh_node(B),
  A != B,
  reachable(A, B),
  not mesh_adjacent(A, B).

warn("mesh has direct route between non-adjacent nodes", { from: A, to: B }) :-
  mesh_non_adjacent_pair(A, B),
  R = format("rt-%s-%s", A, B),
  want(net.route, R),
  not mesh_allow_direct_route(A, B).

# --- Auto-mesh demo (derived links from a list) ---
#
# The mesh itself is described as a *list setting* for human ergonomics.
# We immediately explode it into row facts, then derive edges, then rely on
# reachability closure to fill in the implied connectivity.

mesh_vpc(Name, Net, I) :-
  env(Env),
  setting(Env, .mesh.vpcs, Vpcs),
  member(Vpcs, I, { name: Name, net: Net }).

resource net.vpc Name {
  cidr = Net
  tags = { segment: mesh }
} :-
  mesh_vpc(Name, Net, _I).

# Derive a sparse base graph: connect consecutive VPCs in the list.
# The reachability closure will infer core_a -> core_c via core_b.
mesh_peer(A, B) :-
  mesh_vpc(A, NetA, IA),
  mesh_vpc(B, NetB, IB),
  IB = IA + 1,
  not inet_overlaps(NetA, NetB).

# Feed the derived edges into the global peering/link graph.
peer(A, B) :- mesh_peer(A, B).

# Optional: materialize the derived peerings as explicit peering resources.
resource net.vpc_peering P {
  requester_vpc_id = ref(net.vpc, A, .id)
  accepter_vpc_id = ref(net.vpc, B, .id)
} :-
  mesh_peer(A, B),
  P = format("mesh-peer-%s-%s", A, B).

# Policy 1: peered VPCs must not overlap.
deny("peered vpcs must not overlap", { a: A, b: B }) :-
  peer(A, B),
  arg(net.vpc, A, .cidr, NetA),
  arg(net.vpc, B, .cidr, NetB),
  inet_overlaps(NetA, NetB).

# Policy 2: tenant isolation.
# Only enforced in prod, and depends on the reachability closure.
deny("tenant isolation violated", { a: A, b: B }) :-
  env(prod),
  tenant(A, _),
  tenant(B, _),
  A != B,
  reachable(A, B).

# Fixpoint 2: nested group membership.
# This is another common closure problem in real org infra.
group_edge(sre, admins).
group_edge(oncall, sre).

group_member(alice, oncall).
group_member(bob, sre).

in_group(U, G) :- group_member(U, G).
in_group(U, G2) :- in_group(U, G1), group_edge(G1, G2).

warn("user is effectively admin (via group nesting)", { user: U }) :-
  in_group(U, admins).

* Notes

** Lowering, where it is not obvious

- =tenant(t, i) if some i, t in settings[env].tenants= is
  =env(E), setting(E, .tenants, Ts), member(Ts, I, T)=: Rego's =some k, v in=.
- =resource net.vpc t { for tenant(t, i), corp_net(corp) ... }=: =t= in the
  header is a variable because the =for= clause binds it (G-9);
  =resource net.vpc shared= is a static name because nothing binds =shared=.
- =if dest = net.vpc[b].cidr= binds in the clause, so it is a read now:
  =attr(net.vpc, B, cidr, Dest)=. =destination = net.vpc[b].cidr= written as
  a field would have been =ref(net.vpc, B, .cidr)=, a reference, and printed
  as one (G-6). The current program reads, so the rewrite reads.
- ="rt-{a}-{b}" in net.route= is =R = format("rt-%s-%s", A, B), want(net.route, R)=.
- =some i, { name: name, net: cidr } in settings[env].mesh.vpcs= is today's
  =member(Vpcs, I, {name: Name, net: Net})= with the pattern kept. The
  variable is =cidr=, not =net=: =net= is a type namespace (=net.vpc=) and a
  variable may not take its name (G-4).
- =inet_overlaps(net.vpc[a].cidr, net.vpc[b].cidr)= is two =attr= reads:
  today's =arg(net.vpc, A, .cidr, NetA)= in a body is already =attr=.

** Recursion

The two closures read exactly as before, less capitals and the =:-=:

#+begin_src dform
reachable(a, b) if link(a, b)
reachable(a, c) if reachable(a, b), link(b, c)

in_group(u, g) if group_member(u, g)
in_group(u, g2) if in_group(u, g1), group_edge(g1, g2)
#+end_src

Recursion is over relations, and relations stay relations: the dot surface
never enters a recursive rule here, and should not. A recursive rule that
reads an attribute (=adv7_mutual_recursion_attr.df=) is rejected by
stratification whatever it is spelled as; the dot form makes such a cycle
harder to see (=r.next.cidr= is two joins), which is an LSP job (section 9
of the proposal).

** Reads better

- The rules that were already relational lose only capitals and =:-=.
- =resource net.vpc_peering "mesh-peer-{a}-{b}"= puts the name where
  Terraform readers look for it.
- Denies with =inet_overlaps(net.vpc[a].cidr, net.vpc[b].cidr)=.

** Reads worse

- Thirty symbols became strings (tenant, mesh and group names). The file is
  mostly data, so the token count barely moves (-17%).
- =dest= in the route block is a variable that exists only to force a read
  instead of a reference. The Prolog form made the same choice with a
  predicate name (=arg= versus =ref=); the new form makes it with a
  position, which a reader has to know.
